    <script>
        let currentErrors = [];

        async function fetchErrors() {
            const res = await fetch('/api/errors');
            const data = await res.json();
            currentErrors = data;
            const tbody = document.getElementById('errorTableBody');
            tbody.innerHTML = '';
            let criticals = 0;
            data.forEach(item => {
                if(item.level === 'Kritik') criticals++;
                const tr = document.createElement('tr');
                const levelClass = item.level === 'Kritik' ? 'level-kritik' : 'level-uyari';
                tr.innerHTML = `<td>#${item.id}</td><td>${item.project}</td><td class="${levelClass}">${item.level}</td><td style="color: #fff;">${item.message}</td><td style="color: #8b949e;">${item.url}</td><td style="color: #8b949e;">${item.time}</td>`;
                tr.onclick = () => openModal(item.id);
                tbody.appendChild(tr);
            });
            document.getElementById('totalCount').innerText = data.length;
            document.getElementById('criticalCount').innerText = criticals;
        }

        function openModal(id) {
            const err = currentErrors.find(e => e.id === id);
            if(!err) return;
            document.getElementById('modalTitle').innerText = 'Hata İnceleme #' + err.id + ' (' + err.project + ')';
            document.getElementById('modalBody').innerHTML = `
                <div class="modal-row"><span class="modal-label">Seviye:</span> <span class="${err.level === 'Kritik' ? 'level-kritik' : 'level-uyari'}">${err.level}</span></div>
                <div class="modal-row"><span class="modal-label">Endpoint:</span> <span style="color:#fff;">${err.url}</span></div>
                <div class="modal-row"><span class="modal-label">Kullanıcı IP:</span> <span>${err.ip}</span></div>
                <div class="modal-row"><span class="modal-label">Tarayıcı/OS:</span> <span>${err.browser}</span></div>
                <div><span class="modal-label">Stack Trace (Teknik Detay):</span><div class="stack-trace">${err.stack}</div></div>
            `;
            document.getElementById('resolveBtn').onclick = () => resolveError(err.id);
            document.getElementById('errorModal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('errorModal').style.display = 'none';
        }

        async function resolveError(id) {
            await fetch('/api/errors/' + id, { method: 'DELETE' });
            closeModal();
            fetchErrors();
        }

        async function sendTestError() {
            await fetch('/api/errors', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    project: "E-Ticaret Pro",
                    level: "Kritik",
                    message: "NullPointerException: Cannot invoke method on null object",
                    stack: "NullPointerException: at PaymentController.pay (Payment.java:54)",
                    url: "/checkout/pay"
                })
            });
            fetchErrors();
        }

        fetchErrors();
        setInterval(fetchErrors, 3000);
    </script>
